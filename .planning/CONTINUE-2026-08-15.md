## 부록 — 학습 래퍼 내용 (`/workspace/_run_train.sh`, 볼륨 유실 시 재작성용)

```bash
#!/bin/bash
source /workspace/aws_env.sh
source <(sed -n "3,34p" /workspace/start_server.sh)
export PHASE22_BELLE_GREENLIGHT=1
export GEMINI_KEY_PARAM=/sunity/motion/gemini-api-key
# venv CUDA 라이브러리를 베이스보다 먼저 (베이스 cudnn 을 물면 undefined symbol)
NV=/workspace/train_venv/lib/python3.11/site-packages/nvidia
LP=""; for d in $(find "$NV" -maxdepth 2 -name lib -type d 2>/dev/null); do LP="${LP}${d}:"; done
export LD_LIBRARY_PATH="${LP}${LD_LIBRARY_PATH}"
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True
cd /workspace/SunityMotion || exit 1
R=/workspace/SunityMotion/backend/training/sft/run_retrain_cycle.sh
echo "[chain] train 시작 $(date -u +%H:%M:%S) | nvidia lib dirs=$(echo "$LD_LIBRARY_PATH" | tr ':' '\n' | grep -c nvidia)"
bash "$R" train   || { echo "[chain] train 실패 rc=$?"; exit 1; }
echo "[chain] gates 시작 $(date -u +%H:%M:%S)"
bash "$R" gates   || { echo "[chain] gates 실패 rc=$?"; exit 2; }
echo "[chain] promote 시작 $(date -u +%H:%M:%S)"
bash "$R" promote || { echo "[chain] promote 실패 rc=$?"; exit 3; }
echo "[chain] 전 단계 완료 $(date -u +%H:%M:%S)"
```

★래퍼를 원격에서 `sed` 로 고치지 말 것 — 이스케이프가 뭉개져 CUDA 경로가 조용히
깨진다(08-14 실측: `nvidia lib dirs=2` 가 신호, 정상은 18). 파일을 통째로 만들어
`scp` 로 보낼 것.
