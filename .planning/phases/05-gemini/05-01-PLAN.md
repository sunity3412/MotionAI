    구조 (PATTERNS.md §Plan 5-01 박제 패턴 따라):
    ```python
    """GeminiTechniqueRecognizer — TechniqueRecognizer Protocol 의 Gemini 어댑터.
    
    Plan 5-01 (2026-06-04). Plan 01-13 spike `GeminiMomentExtractor` 를 wrap.
    
    박제 정신:
    - D-04: 1회 호출 → 기술명 + 4단계 라벨 + timestamp
    - D-05: v1 채점 = hold moment 라벨만 활성. setup/peak/release = Firestore 박제만
    - D-08: Gemini = 라벨러만, yaml 수치 (정은지 reference 측정값) source 박제 보호
    - D-09: 3-case fallback (API 실패 / Low conf / 미등록)
    - D-13: model = gemini-3.1-pro 단일
    - D-16: lazy import (google.genai / boto3)
    - [[analysis-objectivity-no-human-scores]] reject patterns 2차 가드 박제
    """
    
    from __future__ import annotations
    
    import logging
    from dataclasses import dataclass, field
    from typing import Callable, Iterable
    
    from .technique import (
        JOINT_BENT_OK,
        JOINT_EXTEND,
        FallbackRecognizer,
        TechniqueProfile,
        TechniqueRecognizer,
    )
    from .skeleton import JOINT_KEYS
    
    # 박제 5영상 scope (D-01 박제)
    REGISTERED_MOTIONS: frozenset[str] = frozenset({
        "ref-climb", "ref-foxtop", "ref-foxtop-split", "ref-invert", "ref-sideway-spin",
    })
    
    log = logging.getLogger(__name__)
    
    
    @dataclass
    class GeminiTechniqueRecognizer:
        """Gemini multimodal video → TechniqueProfile (4단계 라벨 + 기술명)."""
        
        extractor: object | None = None  # GeminiMomentExtractor (lazy)
        cache: object | None = None  # TechniqueCache (Plan 5-02 신설, default None = no-cache)
        fallback: TechniqueRecognizer | None = None
        low_confidence_threshold: float = 0.5  # D-10 박제 후 갱신 (5영상 sweep 실측 후)
        motion_query_hint: str | None = None  # 박제 = None (Gemini 자체 분류), 또는 caller 가 hint
        unregistered_hook: Callable[[str, str], None] | None = None  # D-09 case 3 Firestore trigger (Plan 5-02 wiring)
        
        def __post_init__(self) -> None:
            if self.fallback is None:
                self.fallback = FallbackRecognizer()
        
        def recognize(self, angles, frames=None) -> TechniqueProfile:
            """3-case fallback + reject patterns 2차 가드 + joint_expectations 빌드."""
            # Step 1: cache hit
            cached = None
            if self.cache is not None and frames is not None:
                cached = self.cache.lookup(frames)  # frames = video_path
            if cached is not None:
                log.info("GeminiTechniqueRecognizer cache hit")
                return self._profile_from_cache(cached)
            
            # Step 2: extractor lazy init
            if self.extractor is None:
                from sunity_shared.judging import GeminiMomentExtractor
                self.extractor = GeminiMomentExtractor()
            
            # Step 3: Gemini 호출 (D-09 case 1 fallback)
            try:
                response_text, moments, raw_motion_name = self._call_extractor(frames)
            except (RuntimeError, ValueError) as exc:
                log.warning("Gemini API 실패 — FallbackRecognizer 위임: %s", exc)
                profile = self.fallback.recognize(angles, frames=frames)
                # category override → "api_failure" 박제
                return _override_category(profile, "api_failure")
            
            # Step 4: reject patterns 2차 가드 (D-08 박제 정신, [[analysis-objectivity-no-human-scores]])
            from sunity_shared.judging.gemini_moment_extractor import _enforce_no_coordinate_or_score
            _enforce_no_coordinate_or_score(response_text, context="GeminiTechniqueRecognizer.recognize")
            
            # Step 5: low confidence (D-09 case 2)
            mean_conf = sum(m.confidence for m in moments) / len(moments) if moments else 0.0
            if mean_conf < self.low_confidence_threshold:
                log.info("Gemini low conf=%.2f < %.2f — Page 9 단독 채점 path",
                         mean_conf, self.low_confidence_threshold)
                return TechniqueProfile(
                    name="신뢰도 낮음",
                    category="low_confidence",
                    joint_expectations={},
                    requires_hold=True,
                    is_symmetric=False,
                )
            
            # Step 6: motion 정규화 + 미등록 (D-09 case 3)
            canonical, scope_status = self._classify_motion(raw_motion_name)
            if scope_status == "unregistered":
                log.info("Gemini motion 미등록='%s' — Page 9 단독 + 자동 수집 trigger", raw_motion_name)
                if self.unregistered_hook is not None:
                    try:
                        # B3 fix (2026-06-04 revision): hook 에 video_hash 전달 (video_path 박제 X — TERM-DATA-01 schema 정합 + PII 미노출)
                        # compute_video_hash 산출은 adapter 가 책임 (TechniqueCache 도 같은 함수 사용)
                        from sunity_shared.analysis.technique_cache import compute_video_hash
                        video_hash = compute_video_hash(frames) if frames else ""
                        self.unregistered_hook(raw_motion_name, video_hash)
                    except Exception:
                        log.exception("unregistered_hook 실패 (분석은 계속)")
                return TechniqueProfile(
                    name=f"미등록: {raw_motion_name}",
                    category="unregistered",
                    joint_expectations={},
                    requires_hold=True,
                    is_symmetric=False,
                )
            
            # Step 7: 정상 — KeyMoment → joint_expectations 변환 (D-05 hold 라벨만 활성)
            profile = self._build_profile(canonical, moments)
            
            # Step 8: cache store
            if self.cache is not None and frames is not None:
                self.cache.store(frames, {
                    "motion": canonical,
                    "moments": [_serialize_moment(m) for m in moments],
                    "joint_expectations": profile.joint_expectations,
                    "model": "gemini-3.1-pro",
                })
            
            return profile
        
        def _call_extractor(self, frames):
            """extractor 호출 + 응답 raw text + KeyMoment list + raw motion name 반환."""
            # NOTE: extractor.extract_key_moments() 가 raw text 반환 X (현 API 는 list[KeyMoment]).
            # Gemini 응답 raw text 박제는 extractor 가 내부 _last_raw_response 박제 필요 (별도 박제 작업)
            # 본 task 에선 extractor 호출 + raw_response_text 접근 path 박제 + motion_query hint
            moments = self.extractor.extract_key_moments(
                video_uri=frames,
                motion=self.motion_query_hint or "auto",  # auto = Gemini 자체 분류
            )
            # raw 응답 + motion name 은 extractor 내부 attribute 박제 (Plan 5-01 박제 spike 결과 정합)
            raw_text = getattr(self.extractor, "_last_raw_response", "") or ""
            raw_motion_name = getattr(self.extractor, "_last_motion_name", "") or "auto"
            return raw_text, moments, raw_motion_name
        
        def _classify_motion(self, raw_name: str) -> tuple[str, str]:
            """production 정규화 path 박제 — 정확/한국어 alias/부분 매치.
            
            B2 fix (2026-06-04 revision): production 모듈 gemini_motion_classifier.classify_motion_name
            을 호출. spike 는 production 을 import (역방향). Pod/Lambda runtime ModuleNotFoundError 0.
            """
            from .gemini_motion_classifier import classify_motion_name
            return classify_motion_name(raw_name)
        
        def _build_profile(self, motion: str, moments: list) -> TechniqueProfile:
            """KeyMoment[hold] → joint_expectations dict.
            
            D-08 박제 — Gemini = 라벨러만. yaml 의 hold_moment criteria 가 등재된 관절 = EXTEND,
            미등재 = BENT_OK. ref-climb yaml = hold_moment 빈 list → 6관절 모두 BENT_OK.
            정은지 reference 측정값 (Plan 5-00 박제) 가 yaml angle_target 박제 source.
            """
            from sunity_shared.judging.loader import load_grouped_criteria
            criteria_by_moment = load_grouped_criteria(motion)
            hold_criteria = criteria_by_moment.get("hold_moment", [])
            extend_joints = {c.joint_key for c in hold_criteria}
            expectations: dict[str, str] = {}
            for joint_key in JOINT_KEYS:
                expectations[joint_key] = JOINT_EXTEND if joint_key in extend_joints else JOINT_BENT_OK
            return TechniqueProfile(
                name=motion,
                category="recognized",
                joint_expectations=expectations,
                requires_hold=True,
                is_symmetric=False,
            )
        
        def _profile_from_cache(self, cached: dict) -> TechniqueProfile:
            """cache hit 시 dict → TechniqueProfile 복원."""
            return TechniqueProfile(
                name=cached.get("motion", "cached"),
                category="recognized",
                joint_expectations=cached.get("joint_expectations", {}),
                requires_hold=True,
                is_symmetric=False,
            )
    
    
    def _override_category(profile: TechniqueProfile, category: str) -> TechniqueProfile:
        """dataclass 박제 — fallback 위임 결과의 category 만 갱신."""
        from dataclasses import replace
        return replace(profile, category=category)
    
    
    def _serialize_moment(m) -> dict:
        return {
            "moment_key": m.moment_key,
            "timestamp_seconds": m.timestamp_seconds,
            "confidence": m.confidence,
            "frame_index": getattr(m, "frame_index", 0),
        }
    ```
