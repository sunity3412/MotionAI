    ```python
    """TechniqueCache — 영상 hash 기반 Gemini 결과 캡싱.
    
    Plan 5-02 (2026-06-04). D-14 박제 — 같은 영상 재분석 시 Gemini 호출 0.
    belle 시연 + Plan 23 sweep 재실행 비용/지연 0 효과.
    
    박제 정신:
    - D-14: 영상 hash 캡싱 (SHA256 권장, S3 ETag fallback)
    - D-16: lazy import (firebase-admin)
    - [[firestore-nested-array-flat]] — KeyMoment list = flat dict array 박제
    - Open Question 4 (RESEARCH.md): yaml_version 박제 → yaml 갱신 시 mismatch = cache miss
    """
    
    from __future__ import annotations
    
    import hashlib
    import logging
    from dataclasses import dataclass, field
    from pathlib import Path
    from typing import Any
    
    log = logging.getLogger(__name__)
    
    # Plan 5-00 박제 yaml SHA256 — 박제 yaml 갱신 시 자동 stale invalidate.
    # 실제 값은 cache 진입 시점에 동적 계산 (yaml 5개 SHA256 합쳐 hash).
    # 박제 정신: yaml = 정은지 reference 측정값 (분기 2 path) 박제 source.
    
    
    def compute_video_hash(video_path: str | Path, *, chunk_size: int = 8192) -> str:
        """영상 SHA256 (stdlib hashlib). belle 시연 + sweep 재실행 캡싱 key.
        
        S3 ETag fallback (multipart upload 시 non-hex 우려) — Pod 가 영상 다운로드 후
        SHA256 가 가장 안전 + 영상 정합 검증 효과 추가.
        """
        path = Path(video_path)
        if not path.exists():
            raise FileNotFoundError(f"Video path missing: {path}")
        h = hashlib.sha256()
        with path.open("rb") as fp:
            for chunk in iter(lambda: fp.read(chunk_size), b""):
                h.update(chunk)
        return h.hexdigest()
    
    
    # B4 fix (2026-06-04 revision): 절대 경로 박제 + FileNotFoundError 명시 처리 + truncation 제거.
    # 박제 사유: 상대 경로 = CWD 의존 (Pod uvicorn cwd 와 Lambda cwd 다름) → yaml 발견 실패 시 무성 빈 hash
    # 박제 → cache 영구 stale-stable (D-14 invalidation 무효화). 절대 경로 + 명시 raise 가 D-14 박제 정합.
    
    _YAML_CRITERIA_DIR = (
        Path(__file__).resolve().parents[4] / "judging_data" / "criteria"
    )
    # parents[N] 경로 박제 = backend/shared/python/sunity_shared/analysis/technique_cache.py 기준
    #   parents[0] = analysis/, parents[1] = sunity_shared/, parents[2] = python/, parents[3] = shared/, parents[4] = backend/
    # backend/judging_data/criteria/*.yaml 박제 path.
    # **Task 1 작업 시 `python -c "from pathlib import Path; p=Path('backend/shared/python/sunity_shared/analysis/technique_cache.py').resolve(); print(p.parents[4])"` 확인 후 박제 (반환 = backend/ 디렉터리 절대 경로)**
    
    _YAML_FILENAMES = (
        "ref-climb.yaml",
        "ref-foxtop.yaml",
        "ref-foxtop-split.yaml",
        "ref-invert.yaml",
        "ref-sideway-spin.yaml",
    )
    
    
    def compute_yaml_version(*, strict: bool = True) -> str:
        """yaml 5개 (criteria/*.yaml) SHA256 합쳐 박제.
        
        Plan 5-00 박제 yaml 갱신 시 자동 변경 → cache invalidation.
        B4 fix (2026-06-04 revision):
        - 절대 경로 박제 (`Path(__file__).resolve().parents[N] / judging_data / criteria / ...`)
        - strict=True (default): FileNotFoundError 시 raise (무성 누락 처리 0)
        - strict=False: FileNotFoundError 시 logged 경고 + cache disable (반환값 = sentinel string "yaml-missing-cache-disabled")
        - truncation 제거: 64자 full hex 박제 (32+ chars 유지, 16자 박제 cli 우려 0)
        """
        h = hashlib.sha256()
        missing: list[str] = []
        for filename in _YAML_FILENAMES:
            yaml_path = _YAML_CRITERIA_DIR / filename
            if not yaml_path.exists():
                if strict:
                    raise FileNotFoundError(
                        f"yaml criteria missing (B4 fix: cache invalidation 정합 보호 위해 strict raise): {yaml_path}"
                    )
                missing.append(filename)
                continue
            with yaml_path.open("rb") as fp:
                h.update(fp.read())
        if missing:
            log.warning("compute_yaml_version: yaml %d개 누락 (strict=False, cache disabled): %s", len(missing), missing)
            return "yaml-missing-cache-disabled"
        return h.hexdigest()  # full 64 chars (B4 fix: truncation 제거)
    
    
    @dataclass
    class TechniqueCache:
        """Firestore gemini_cache/{hash} 컬렉션 (전역 공유) + in-memory layer 2단.
        
        in-memory = Pod 단일 분석 내 중복 호출 흡수.
        Firestore = Pod 재기동/belle 시연/sweep 재실행 영구 캡싱.
        """
        
        model_name: str = "gemini-3.1-pro"  # D-13 박제
        yaml_version: str = field(default_factory=compute_yaml_version)
        _memory: dict[str, dict] = field(default_factory=dict, init=False, repr=False)
        
        def lookup(self, video_path: str | Path) -> dict | None:
            """video_path → hash → cache lookup.
            
            반환 dict schema (D-14 + Open Question 4):
            {
              "motion": "ref-foxtop",
              "moments": [{"moment_key": "hold", "timestamp_seconds": 5.5, "frame_index": 49, "confidence": 0.88}, ...],
              "joint_expectations": {"left_shoulder": "extend", ...},
              "model": "gemini-3.1-pro",
              "yaml_version": "abc123...",
              "cache_hit": True
            }
            """
            try:
                video_hash = compute_video_hash(video_path)
            except FileNotFoundError:
                log.warning("Video path 없음 — cache lookup skip")
                return None
            cache_key = (video_hash, self.model_name, self.yaml_version)
            
            # Step 1: in-memory hit
            mem_key = self._mem_key(cache_key)
            if mem_key in self._memory:
                log.debug("TechniqueCache in-memory hit: %s", video_hash[:8])
                return dict(self._memory[mem_key])
            
            # Step 2: Firestore lookup (lazy import)
            try:
                from sunity_shared import firestore_admin
                doc = firestore_admin.get_gemini_cache(video_hash)
            except Exception as exc:
                log.warning("Firestore gemini_cache lookup 실패: %s", exc)
                return None
            
            if doc is None:
                return None
            
            # yaml_version + model 정합 검증
            if doc.get("yaml_version") != self.yaml_version:
                log.info("cache yaml_version mismatch (stale) — invalidate: %s vs %s",
                         doc.get("yaml_version"), self.yaml_version)
                return None
            if doc.get("model") != self.model_name:
                log.info("cache model mismatch — invalidate")
                return None
            
            # in-memory 갱신
            self._memory[mem_key] = dict(doc)
            log.info("TechniqueCache Firestore hit: %s", video_hash[:8])
            return dict(doc)
        
        def store(self, video_path: str | Path, gemini_result: dict) -> None:
            """video_path → hash → Firestore + in-memory store.
            
            gemini_result 박제 시 yaml_version + model_name 자동 박제.
            """
            try:
                video_hash = compute_video_hash(video_path)
            except FileNotFoundError:
                log.warning("Video path 없음 — cache store skip")
                return
            
            doc = dict(gemini_result)
            doc["yaml_version"] = self.yaml_version
            doc["model"] = self.model_name
            doc["video_hash"] = video_hash
            
            # nested-array 금지 정합 검증 ([[firestore-nested-array-flat]])
            if "moments" in doc and doc["moments"]:
                for m in doc["moments"]:
                    if not isinstance(m, dict):
                        raise TypeError(
                            f"moments entry must be flat dict (firestore-nested-array-flat 박제): got {type(m)}"
                        )
                    for k, v in m.items():
                        if isinstance(v, (list, tuple)):
                            raise TypeError(
                                f"moments[{k}] must be scalar (firestore nested array 금지): got {type(v)}"
                            )
            
            cache_key = (video_hash, self.model_name, self.yaml_version)
            self._memory[self._mem_key(cache_key)] = dict(doc)
            
            try:
                from sunity_shared import firestore_admin
                firestore_admin.store_gemini_cache(video_hash, doc)
            except Exception as exc:
                log.warning("Firestore gemini_cache store 실패 (in-memory 만 유효): %s", exc)
        
        @staticmethod
        def _mem_key(key: tuple[str, str, str]) -> str:
            return ":".join(key)
    ```
