          tree = ast.parse(src)
          strings: list[tuple[str, int]] = []
          for node in ast.walk(tree):
              if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name) and node.targets[0].id in ("_COPY_TEMPLATES", "_MODE_PREFIX"):
