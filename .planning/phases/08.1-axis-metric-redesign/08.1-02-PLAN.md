```python
#!/usr/bin/env python3
"""Phase 8.1 sweep CLI — 정은지 reference 5 영상 재sweep.

Phase 8 의 Pod 임시 스크립트 (08-03-SUMMARY.md Deviations) 의 정제 버전.
mode='self' (mode3), sweep_temp/ prefix (SQS 우회), pipeline._process direct call.
"""
import argparse
import importlib.util
import sys
import time
import uuid
from pathlib import Path
import boto3

BACKEND = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(BACKEND / "shared" / "python"))

from sunity_shared import firestore_admin as fa, models  # noqa: E402

spec = importlib.util.spec_from_file_location(
    "sunity_pipeline_app", BACKEND / "functions" / "pipeline" / "app.py"
)
pipeline = importlib.util.module_from_spec(spec)
spec.loader.exec_module(pipeline)
pipeline._ensure_adapters()


def main() -> int:
    parser = argparse.ArgumentParser(description="Phase 8.1 sweep CLI")
    parser.add_argument(
        "--sweep-uid", required=True,
        help="Unique sweep UID, e.g. sweep_phase8_1_<timestamp>",
    )
    parser.add_argument(
        "--videos",
        default="ref-invert,ref-climb,ref-foxtop,ref-foxtop-split,ref-sideway-spin",
        help="Comma-separated reference video names (under reference/ prefix)",
    )
    parser.add_argument("--bucket", default="sunity-motion-pilot-videos")
    parser.add_argument("--dry-run", action="store_true")
    args = parser.parse_args()

    s3 = boto3.client("s3")
    db = fa._db()
    videos = [v.strip() for v in args.videos.split(",") if v.strip()]

    print(f"[setup] sweep_uid={args.sweep_uid} videos={len(videos)}", flush=True)
    if args.dry_run:
        print("[dry-run] no S3 copy / no Firestore write / no _process", flush=True)
        return 0

    for i, name in enumerate(videos, start=1):
        analysis_id = uuid.uuid4().hex
        new_key = f"sweep_temp/{args.sweep_uid}/{analysis_id}.mp4"
        print(f"\n[sweep {i}/{len(videos)}] {name} -> {analysis_id}", flush=True)

        # 1. Copy S3
        t0 = time.time()
        s3.copy_object(
            Bucket=args.bucket,
            CopySource={"Bucket": args.bucket, "Key": f"reference/{name}.mp4"},
            Key=new_key,
        )
        print(f"  [{time.time()-t0:.1f}s] copied to {new_key}", flush=True)

        # 2. Firestore doc
        doc_ref = (
            db.collection("users").document(args.sweep_uid)
            .collection("analyses").document(analysis_id)
        )
        doc_ref.set({
            "analysisId": analysis_id,
            "uid": args.sweep_uid,
            "mode": models.MODE_SELF,
            "status": models.STATUS_QUEUED,
            "videoKey": new_key,
            "videoFormat": "mp4",
            "fileSizeBytes": 0,
            "createdAt": time.time(),
            "sourceLabel": f"sweep_phase8_1:{name}",
        })

        # 3. _process direct call
        t1 = time.time()
        try:
            pipeline._process(args.bucket, new_key, args.sweep_uid, analysis_id)
            print(f"  [{time.time()-t1:.1f}s] _process OK", flush=True)
        except Exception as exc:  # noqa: BLE001
            print(f"  _process FAILED: {type(exc).__name__}: {exc}", flush=True)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```
