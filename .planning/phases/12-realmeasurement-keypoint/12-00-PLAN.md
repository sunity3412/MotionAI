def _build_keypoints_2d_from_rtmw(
    raw_133: np.ndarray,
    img_w: int,
    img_h: int,
) -> dict[str, Keypoint2D]:
    """RTMW 133 keypoints → COCO-17 2D normalized [0, 1].

    R1 fix (Codex 직접 리뷰 2026-06-10): 기존 None 반환 → 실 좌표 박제.
    Phase 12 KeypointOverlay 의 데이터 source.
    """
    if img_w <= 0 or img_h <= 0:
        return {}
    result: dict[str, Keypoint2D] = {}
    for coco_name, rtmw_idx in RTMW_133_TO_COCO17.items():
        kp = raw_133[rtmw_idx]  # shape (3,) = (x, y, score)
        x, y, score = float(kp[0]), float(kp[1]), float(kp[2])
        result[coco_name] = Keypoint2D(
            x=x / img_w,
            y=y / img_h,
            visibility=score,
            ,
        )
    return result
```

    knee_mid: tuple[float, float] | None  # 무릎 keypoint 누락 시 None


def _midpoint(a: Keypoint2D, b: Keypoint2D) -> tuple[float, float]:
    return ((a.x + b.x) / 2.0, (a.y + b.y) / 2.0)


def compute_axis_frames(pose_frames: list[PoseFrame]) -> list[AxisFrame | None]:
    """각 PoseFrame 의 중심축 폴리라인 산출. R2 (Codex 리뷰 2026-06-10) 정합.

    어깨중심/골반중심 둘 다 있으면 AxisFrame 생성 (무릎중심은 옵션).
    어깨 또는 골반 keypoint 누락 시 None 반환 (외부 fallback 책임).
    """
    result: list[AxisFrame | None] = []
    for frame in pose_frames:
        kp = frame.keypoints_2d
        if not kp:
            result.append(None)
            continue
        try:
            sm = _midpoint(kp["left_shoulder"], kp["right_shoulder"])
            hm = _midpoint(kp["left_hip"], kp["right_hip"])
        except KeyError:
            result.append(None)
            continue
        # knee 옵션
        try:
            km = _midpoint(kp["left_knee"], kp["right_knee"])
        except KeyError:
            km = None
        result.append(AxisFrame(shoulder_mid=sm, hip_mid=hm, knee_mid=km))
    return result
```
