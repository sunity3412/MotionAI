     ```
     function jointConfidence(report: KeypointReport | null | undefined, jointName: string): number | null {
       if (!report) return null;
       const j = report.joints.indexOf(jointName);
       if (j < 0) return null;
       const J = report.joints.length;
       const stride = J;
       let sum = 0, count = 0;
       for (let t = 0; t < report.frames; t++) {
         const v = report.confidence[t * stride + j];
         if (typeof v === 'number' && Number.isFinite(v)) { sum += v; count++; }
       }
       return count > 0 ? sum / count : null;
     }
     ```
